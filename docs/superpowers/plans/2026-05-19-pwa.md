# PWA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make GrowWatch installable on mobile/desktop, offline-capable with last-known sensor data, and wired for push notifications.

**Architecture:** Angular's `@angular/service-worker` (NGSW) handles app-shell caching and installability. Because GraphQL uses POST (not cacheable by NGSW), offline data is handled by writing the last-known sensor reading to `localStorage` in `SensorService` — components fall back to that when `ConnectivityService.isOnline()` is false. Push notification groundwork adds a `PushNotificationService` on the frontend and a `/api/push-subscribe` endpoint + VAPID keys on the backend.

**Tech Stack:** Angular 21 (`@angular/service-worker`, `provideServiceWorker`), NGSW, Web Push API, `web-push` (backend), `sharp` (icon generation, dev-only), `localStorage` for offline fallback.

```javascript
// frontend/scripts/generate-pwa-icons.mjs
import sharp from 'sharp';
import { readFileSync, mkdirSync } from 'fs';
import { resolve, dirname } from 'path';
import { fileURLToPath } from 'url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const svgPath = resolve(__dirname, '../public/favicon.svg');
const outDir = resolve(__dirname, '../public/icons');

mkdirSync(outDir, { recursive: true });

const svg = readFileSync(svgPath);

// Standard icons
for (const size of [192, 512]) {
  await sharp(svg).resize(size, size).png().toFile(`${outDir}/icon-${size}x${size}.png`);
  console.log(`✓ icon-${size}x${size}.png`);
}

// Maskable icon: add 15% padding so the leaf stays in the safe zone
const maskableSize = 512;
const padding = Math.round(maskableSize * 0.15);
const innerSize = maskableSize - padding * 2;
const resized = await sharp(svg).resize(innerSize, innerSize).png().toBuffer();
await sharp({
  create: { width: maskableSize, height: maskableSize, channels: 4, background: '#1a4a35' }
})
  .composite([{ input: resized, gravity: 'center' }])
  .png()
  .toFile(`${outDir}/icon-maskable-512x512.png`);
console.log('✓ icon-maskable-512x512.png');
```

```typescript
import { Injectable } from '@angular/core';
import { fromEvent, merge } from 'rxjs';
import { map, startWith } from 'rxjs/operators';
import { toSignal } from '@angular/core/rxjs-interop';

@Injectable({ providedIn: 'root' })
export class ConnectivityService {
  readonly isOnline = toSignal(
    merge(
      fromEvent(window, 'online').pipe(map(() => true)),
      fromEvent(window, 'offline').pipe(map(() => false)),
    ).pipe(startWith(navigator.onLine)),
    { initialValue: navigator.onLine },
  );
}
```

```typescript
private readonly CACHE_KEY = 'growwatch-last-reading';

cacheLatestReading(data: SensorData): void {

getCachedReading(): SensorData | null {
```

```typescript
// backend/src/push/vapid.ts
import webPush from 'web-push';

export function initVapid() {
  const publicKey  = process.env.VAPID_PUBLIC_KEY;
  const privateKey = process.env.VAPID_PRIVATE_KEY;
  const subject    = process.env.VAPID_SUBJECT;
  if (!publicKey || !privateKey || !subject) {
    console.warn('[push] VAPID keys not set — push notifications disabled');
    return;
  }
  webPush.setVapidDetails(subject, publicKey, privateKey);
  console.log('[push] VAPID initialised');
}

export const vapidPublicKey = () => process.env.VAPID_PUBLIC_KEY ?? '';
```

- [ ] **Create `backend/src/push/push.routes.ts`**

```typescript
// backend/src/push/push.routes.ts
import { Router } from 'express';
import webPush, { PushSubscription } from 'web-push';

const router = Router();

// In-memory store for the POC; replace with MongoDB collection when scaling.
const subscriptions: PushSubscription[] = [];

router.get('/vapid-public-key', (_req, res) => {
  res.json({ key: process.env.VAPID_PUBLIC_KEY ?? '' });
});

router.post('/subscribe', (req, res) => {
  const subscription: PushSubscription = req.body;
  if (!subscription?.endpoint) { res.status(400).json({ error: 'Invalid subscription' }); return; }
  const already = subscriptions.some(s => s.endpoint === subscription.endpoint);
  if (!already) subscriptions.push(subscription);
  res.status(201).json({ ok: true });
});

router.post('/unsubscribe', (req, res) => {
  const { endpoint } = req.body;
  const idx = subscriptions.findIndex(s => s.endpoint === endpoint);
  if (idx !== -1) subscriptions.splice(idx, 1);
  res.json({ ok: true });
});

// Test endpoint — send a push to all subscribers (dev use only)
router.post('/send-test', async (_req, res) => {
  const payload = JSON.stringify({ title: 'GrowWatch', body: 'Test push notification' });
  const results = await Promise.allSettled(
    subscriptions.map(sub => webPush.sendNotification(sub, payload))
  );
  res.json({ sent: results.filter(r => r.status === 'fulfilled').length });
});

export { router as pushRouter };
```

```typescript
    return 'serviceWorker' in navigator && 'PushManager' in window;
  }

  get permission(): NotificationPermission {
    return 'Notification' in window ? Notification.permission : 'denied';
  }

  async subscribe(): Promise<boolean> {
    if (!this.isSupported) return false;
    try {
      const perm = await Notification.requestPermission();
      if (perm !== 'granted') return false;

      const { key } = await this.http.get<{ key: string }>(`${API}/vapid-public-key`).toPromise() ?? { key: '' };
      if (!key) return false;

      const reg = await navigator.serviceWorker.ready;
      const existing = await reg.pushManager.getSubscription();
      const subscription = existing ?? await reg.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: this.urlBase64ToUint8Array(key),
      });

      await this.http.post(`${API}/subscribe`, subscription.toJSON()).toPromise();
      return true;
    } catch (e) {
      console.error('[push] subscribe failed', e);
      return false;
    }
  }

  async unsubscribe(): Promise<void> {
    if (!this.isSupported) return;
    const reg = await navigator.serviceWorker.ready;
    const sub = await reg.pushManager.getSubscription();
    if (!sub) return;
    await this.http.post(`${API}/unsubscribe`, { endpoint: sub.endpoint }).toPromise();
    await sub.unsubscribe();
  }

  private urlBase64ToUint8Array(base64String: string): Uint8Array {
    const padding = '='.repeat((4 - (base64String.length % 4)) % 4);
    const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/');
    const rawData = atob(base64);
    return Uint8Array.from([...rawData].map(c => c.charCodeAt(0)));
  }
}
```
